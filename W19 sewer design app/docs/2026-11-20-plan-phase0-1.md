> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (inline, in this session: the engineer

**Goal:** A local sewer design app whose engine reproduces W18's SewerGEMS-matched design of Ibri on every pipe, then an
interface and an AI assistant over it.

**Architecture:** Phase 0 copies W18's scripts into W19 with their paths moved into one environment variable, and
proves they still give W18's answer from frozen fixtures. Phase 1A rebuilds the flow, sizing and analysis steps as a
clean vectorised library. The library drives the ported chain (split, naming, products) and is gated on all 12 cases
and on SewerGEMS's own export. Phase 1B puts FastAPI, a browser interface and the assistant on top.

**Tech Stack:** Python 3.12, numpy 2.4.6, pandas 3.0.3, geopandas 1.1.4, shapely 2.1.2, pytest; phase 1B adds
FastAPI 0.165.0, uvicorn 0.49.0, anthropic 2.12.0, React + TypeScript + Vite, MapLibre GL JS, AG Grid Community,
Playwright.

**Spec:** `W19 sewer design app/docs/2026-11-20-sewer-design-app-design.md`

```python
# sewerapp/fixtures.py
"""The frozen reference files (tests/fixtures/): W18's inputs and outputs and SewerGEMS's exports, copied once by
tools/freeze_fixtures.py. A file may be stored gzipped; callers name it without the .gz."""
import gzip, json, os, shutil
import pandas as pd

FIX = os.path.join(os.path.dirname(os.path.dirname(os.path.abspath(__file__))), "tests", "fixtures")


def path(rel):
    p = os.path.join(FIX, rel)
    return p if os.path.exists(p) else p + ".gz"


def read_csv(rel, **kw):
    return pd.read_csv(path(rel), **kw)


def materialize(rel, dest):
    os.makedirs(os.path.dirname(dest), exist_ok=True)
    src = path(rel)
    if src.endswith(".gz"):
        with gzip.open(src, "rb") as f, open(dest, "wb") as g: shutil.copyfileobj(f, g)
    else:
        shutil.copyfile(src, dest)
    return dest


def manifest():
    return json.load(open(os.path.join(FIX, "MANIFEST.json"), encoding="utf-8"))
```

```python
def frame_differences(name, a, b, rtol=1e-9, atol=1e-9):
    out = []
    if list(a.columns) != list(b.columns): return [f"{name}: columns {list(a.columns)} != {list(b.columns)}"]
    if len(a) != len(b): return [f"{name}: {len(a)} rows != {len(b)}"]
    for col in a.columns:
        x, y = a[col].to_numpy(), b[col].to_numpy()
        if np.issubdtype(x.dtype, np.number) and np.issubdtype(y.dtype, np.number):
            bad = ~np.isclose(x.astype(float), y.astype(float), rtol=rtol, atol=atol, equal_nan=True)
        else:
            bad = (pd.Series(x).astype(str) != pd.Series(y).astype(str)).to_numpy()
        if bad.any():
            i = int(np.flatnonzero(bad)[0])
            out.append(f"{name}: column {col}, {int(bad.sum())} rows differ, first row {i}: {x[i]!r} != {y[i]!r}")
    return out


def json_differences(name, a, b, rtol=1e-9, atol=1e-9, where=""):
    if isinstance(a, dict) and isinstance(b, dict):
        if set(a) != set(b): return [f"{name}{where}: keys {sorted(set(a) ^ set(b))} differ"]
        return [d for k in a for d in json_differences(name, a[k], b[k], rtol, atol, f"{where}.{k}")]
    if isinstance(a, list) and isinstance(b, list):
        if len(a) != len(b): return [f"{name}{where}: length {len(a)} != {len(b)}"]
        return [d for i, (x, y) in enumerate(zip(a, b)) for d in json_differences(name, x, y, rtol, atol, f"{where}[{i}]")]
    if isinstance(a, (int, float)) and isinstance(b, (int, float)) and not isinstance(a, bool):
        return [] if math.isclose(a, b, rel_tol=rtol, abs_tol=atol) else [f"{name}{where}: {a} != {b}"]
    return [] if a == b else [f"{name}{where}: {a!r} != {b!r}"]


def case_differences(case_dir, case, rtol=1e-9, atol=1e-9):
    out = []
    for f in FILES:
        rel = f.format(c=case)
        out += frame_differences(rel, pd.read_csv(os.path.join(case_dir, rel)),
                                 fixtures.read_csv(f"golden/{case}/{rel}"), rtol, atol)
    for f in JSONS:
        rel = f.format(c=case)
        mine = json.load(open(os.path.join(case_dir, rel), encoding="utf-8"))
        gold = json.load(open(fixtures.path(f"golden/{case}/{rel}"), encoding="utf-8"))
        out += json_differences(rel, mine, gold, rtol, atol)
    return out
```

```python
# sewerapp/engine/hydraulics.py
"""Gravity pipe hydraulics for circular pipes, vectorised. Colebrook-White with ks 1.5 mm for all sizes and nu
1.141e-6 m2/s at 15 degC (PAM-GUD-203 p24, p25, p28). Same arithmetic as W18's scalar functions."""
import numpy as np

G, NU, KS = 9.81, 1.141e-6, 1.5e-3


def section(D, y):
    """Flow area and hydraulic radius of a circular pipe of inside diameter D at depth y."""
    D = np.asarray(D, float); y = np.asarray(y, float)
    th = 2 * np.arccos(1 - 2 * y / D)
    A = D * D / 8 * (th - np.sin(th))
    return A, A / (D * th / 2)


def v_colebrook(R, S, ks=KS, nu=NU):
    a = np.sqrt(8 * G * R * S)
    return -2 * a * np.log10(ks / (14.8 * R) + 2.51 * nu / (4 * R * a))


def q_capacity(D, S, ratio, ks=KS, nu=NU):
    """Flow at depth ratio * D, at gradient S."""
    D = np.asarray(D, float)
    A, R = section(D, np.asarray(ratio, float) * D)
    return A * v_colebrook(R, S, ks, nu)


def normal_depth(Q, D, S, ks=KS, nu=NU, iters=50):
    """Depth ratio y/D and velocity at normal depth, by bisection as W18 (50 halvings between 1e-6 D and 0.999 D)."""
    Q, D, S = np.broadcast_arrays(np.asarray(Q, float), np.asarray(D, float), np.asarray(S, float))
    lo, hi = 1e-6 * D, 0.999 * D
    for _ in range(iters):
        y = (lo + hi) / 2
        A, R = section(D, y)
        over = A * v_colebrook(R, S, ks, nu) > Q
        hi = np.where(over, y, hi); lo = np.where(over, lo, y)
    y = (lo + hi) / 2
    A, _ = section(D, y)
    pos = Q > 0
    return np.where(pos, y / D, 0.0), np.where(pos, Q / np.where(pos, A, 1.0), 0.0)
```

```python
# sewerapp/engine/peaking.py
"""Peaking of sanitary flow. The model's Extreme Flow Setup table is applied to each pipe's cumulative upstream unit
load in L/s, linear between rows (SewerGEMS's way, read off and proved on Ibri SC1-2056)."""
import numpy as np

# Ibri's 'Peltier100-Merriamck' (the model's spelling): Peltier to 1 L/s, Merrimack above, step row 1.001 -> 3.56
IBRI_PELTIER100_MERRIMACK = (
    (0.1, 4.66), (0.15, 4.08), (0.2, 3.74), (0.3, 3.33), (0.5, 2.91), (0.75, 2.65), (1, 2.5), (1.001, 3.56),
    (1.5, 3.39), (2, 3.28), (3, 3.12), (5, 2.93), (7.5, 2.79), (10, 2.7), (15, 2.57), (20, 2.48), (30, 2.36),
    (50, 2.22), (75, 2.11), (100, 2.04), (150, 1.94), (200, 1.88), (300, 1.79), (500, 1.68), (750, 1.6), (1000, 1.54))


class PeakingTable:
    def __init__(self, rows):
        rows = sorted(rows)
        self.q = np.array([r[0] for r in rows], float)
        self.f = np.array([r[1] for r in rows], float)

    def factor(self, q_ls):
        r = np.interp(q_ls, self.q, self.f)
        return float(r) if np.ndim(r) == 0 else r


def peltier(q_ls):
    """PAM-GUD-201 p72: PF = 1.5 + 1 / sqrt(Qm), Qm the average flow in L/s."""
    return 1.5 + 1.0 / np.sqrt(q_ls)


def merrimack(q_ls):
    """PAM-GUD-201 p71 7.4.2: Qpdf = 2.65 Qadf^0.879, flows in Ml/d; returned as the factor Qpdf / Qadf."""
    qadf = np.asarray(q_ls, float) * 86.4 / 1000.0
    return 2.65 * qadf ** -0.121
```

```python
# sewerapp/model/network.py
"""A gravity network as two tables and a topological order. Flow direction is from the higher invert to the lower
(W18's rule: a conduit's start and stop need not follow the flow)."""
from dataclasses import dataclass, field
import numpy as np, pandas as pd


@dataclass
class Network:
    pipes: pd.DataFrame
    nodes: pd.DataFrame
    order: np.ndarray
    terminals: set = field(default_factory=set)

    @classmethod
    def from_w18(cls, conduits, manholes):
        d = conduits[conduits.active == True].copy().reset_index(drop=True)
        hi_start = d.start_inv >= d.stop_inv
        d["up"] = np.where(hi_start, d.start, d.stop); d["dn"] = np.where(hi_start, d.stop, d.start)
        d["nominal_mm"] = d.size_label.str.extract(r"(\d+)")[0].astype(int)
        d["inside_m"] = d.diameter_mm / 1000.0
        d["slope"] = (d[["start_inv", "stop_inv"]].max(axis=1) - d[["start_inv", "stop_inv"]].min(axis=1)) / d.length_m
        d["sub"] = d.label.str.split("-").str[0]
        nodes = manholes.set_index("id")[["label", "x", "y", "ground"]]
        return cls(d, nodes, _order(d), set(d.dn) - set(d.up))


def _order(d):
    """Kahn's order: a pipe once every pipe arriving at its upstream node is placed."""
    arriving = d.groupby("dn").size().to_dict()
    leaving = dict(zip(d.up, d.index))
    ready = [k for k, u in zip(d.index, d.up) if u not in arriving]
    left = dict(arriving); order = []
    while ready:
        k = ready.pop(); order.append(k); n = d.at[k, "dn"]
        if n in left:
            left[n] -= 1
            if left[n] == 0 and n in leaving: ready.append(leaving[n])
    if len(order) != len(d): raise ValueError(f"not a tree: {len(d) - len(order)} pipes unordered")
    return np.array(order)
```

```python
# sewerapp/engine/flows.py
"""Design flow in every pipe (W18's rule, proved against SewerGEMS on all 19,083 Ibri pipes, flows within 0.004 %):
  q = PF(cumulative unit load, L/s) x cumulative unit load (the pipe's upstream manhole included)
      + infiltration x cumulative conduit length (the pipe's own length included)   (G201 p72: 720 L/day per km)
      + transfers pumped in upstream, not re-peaked.
A cut pipe (a lift station) does not pass its loads on: its peaked flow lands on its downstream manhole as a transfer."""
import numpy as np


def accumulate(net, base_m3d, transfer_m3d, peaking, inf_m3d_per_km=0.72, cut=frozenset()):
    p = net.pipes; up = p.up.to_numpy(); dn = p.dn.to_numpy(); L = p.length_m.to_numpy(float)
    n = len(p); cb = np.zeros(n); cl = np.zeros(n); ct = np.zeros(n); q = np.zeros(n)
    acc_b, acc_l, acc_t = {}, {}, dict(transfer_m3d)
    lifts = []
    for k in net.order:
        u = up[k]
        cb[k] = base_m3d.get(u, 0.0) + acc_b.get(u, 0.0)
        cl[k] = L[k] + acc_l.get(u, 0.0)
        ct[k] = acc_t.get(u, 0.0)
        d = dn[k]
        if k in cut:
            q[k] = cb[k] * peaking.factor(cb[k] / 86.4) + cl[k] / 1000.0 * inf_m3d_per_km + ct[k]
            acc_t[d] = acc_t.get(d, 0.0) + q[k]; lifts.append((int(k), int(d), float(q[k])))
        else:
            acc_b[d] = acc_b.get(d, 0.0) + cb[k]; acc_l[d] = acc_l.get(d, 0.0) + cl[k]; acc_t[d] = acc_t.get(d, 0.0) + ct[k]
    q = cb * peaking.factor(cb / 86.4) + cl / 1000.0 * inf_m3d_per_km + ct
    return q, lifts
```

```python
# sewerapp/engine/transfers.py
"""Flows passed between subnetworks. A subnetwork's outfall flow without transfers (W18 read it from SewerGEMS,
bridge/out/w18/f0) is computed here; the cumulative transfer of an outfall = its own flow + everything transferred into
its subnetwork, upstream first (W18 routing.transfers; peaks add, not re-peaked)."""
import collections
from sewerapp.engine import flows


def own_f0(net, base_m3d, peaking, inf_m3d_per_km=0.72):
    q, _ = flows.accumulate(net, base_m3d, {}, peaking, inf_m3d_per_km)
    p = net.pipes; out = {}
    for k in range(len(p)):
        if p.dn.iat[k] in net.terminals:
            out[p["sub"].iat[k]] = out.get(p["sub"].iat[k], 0.0) + q[k]
    return out


def cumulative(routes, f0):
    into = collections.defaultdict(list)
    for s, _, r in routes: into[r].append(s)
    total = {}

    def flow(s):
        if s not in total: total[s] = f0.get(s, 0.0) + sum(flow(u) for u in into[s])
        return total[s]

    for s in f0: flow(s)
    return [(s, mh, total[s]) for s, mh, _ in routes], total
```

```python
# sewerapp/engine/design.py
"""Automated sizing as SewerGEMS does it (proved blind on Ibri SC1-2051, then on 2056): each pipe takes the smallest
catalogue size whose Colebrook-White capacity at the d/D limit carries its flow at the gradient of its CURRENT size,
pass after pass from the modeller's sizes until no size changes. Gradient = the Table 11 minimum of the size."""
import numpy as np
from sewerapp.engine.hydraulics import q_capacity


def dd_limit(nominal):
    """PAM-GUD-203 p27 Table 10: d/D at peak <= 0.65 up to DN355 (inside 313 mm), <= 0.50 above."""
    return np.where(np.asarray(nominal) <= 355, 0.65, 0.50)


def table11_slopes(net, cat_nominal):
    """The gradient per size of the network as designed (every Ibri pipe sits on PAM-GUD-203 p29 Table 11); sizes it does
    not use: 0.075 % from DN900, 0.205 % for DN400 (Table 11). W18 final_design.py lines 41-45."""
    s = net.pipes.groupby("nominal_mm").slope.agg(lambda x: x.round(6).mode().iloc[0]).to_dict()
    for n in cat_nominal:
        if n not in s: s[n] = 0.00075 if n >= 900 else {400: 0.00205}.get(n, np.nan)
    missing = [n for n in cat_nominal if np.isnan(s[n])]
    if missing: raise ValueError(f"no Table 11 gradient for DN{missing}")
    return {int(k): float(v) for k, v in s.items()}


def size_passes(q_m3s, start_nominal, cat_nominal, cat_inside_m, slope_of, max_passes=20):
    cat_nominal = np.asarray(cat_nominal, int); D = np.asarray(cat_inside_m, float)[None, :]
    lim = dd_limit(cat_nominal)[None, :]; q = np.asarray(q_m3s, float)[:, None]
    nom = np.asarray(start_nominal, int).copy()
    for p in range(1, max_passes + 1):
        S = np.array([slope_of[int(n)] for n in nom])[:, None]
        ok = q_capacity(D, S, lim) >= q
        new = cat_nominal[np.where(ok.any(axis=1), ok.argmax(axis=1), len(cat_nominal) - 1)]
        changed = int((new != nom).sum()); nom = new
        if changed == 0: return nom, p
    raise RuntimeError(f"sizing did not settle in {max_passes} passes")
```

```python
"""The design step of one case (scenario x design year) with the clean engine. Writes what W18's final_design.py
wrote, so the ported split, naming and products read it unchanged: <case>/exports/conduits_<case>a.csv (design year),

def design_case(root, name, year, lifts=True, own_f0=False):
    tag = f"{name}-{year}"; R = os.path.join(root, SUB); INP = os.path.join(R, "inputs")
    EXP = os.path.join(R, tag, "exports"); os.makedirs(EXP, exist_ok=True)

                           pd.read_csv(os.path.join(INP, "manholes_network.csv")))

    cat = pd.read_csv(os.path.join(INP, "catalogue.csv")).sort_values("nominal_mm")

    slope_of = design.table11_slopes(net, cat_nom)
    start = w18csv.start_sizes(os.path.join(INP, "conduits_start_sizes.csv"), d.label)
    loads = pd.read_csv(os.path.join(root, "results", "loads", "manhole_loads.csv"), index_col="manhole_id")
    ren = w18csv.read_rename(os.path.join(root, "bridge", "inv", "rename_nodes.csv"))

    cut = set()

    for y in [year] + CHECKS:

        tr = {}
        for s, mh, q in rows:
            tr[ren.new2id[mh]] = tr.get(ren.new2id[mh], 0.0) + q; rows_out.append([int(y), s, ren.new2id[mh], mh, q])
        q, lf = flows.accumulate(net, base, tr, pk, INF, cut)
        Q[y] = pd.Series(q) / 86400.0
        lift_rows += [(int(y), d.label.iat[k], dn, qq) for k, dn, qq in lf]
    nom, passes = design.size_passes(Q[year].values, start, cat_nom, CAT.values, slope_of)

    over = Q[year].values > cap * 1.0001

    up_is_start = (d.up == d.start).values
    e2["start_inv"] = np.where(up_is_start, 1000.0, 1000.0 - drop)
    e2["stop_inv"] = np.where(up_is_start, 1000.0 - drop, 1000.0)
    for f, y in ((f"conduits_{tag}a.csv", year),) + tuple((f"conduits_{tag}-at-{c}.csv", c) for c in CHECKS):
        x = e2.copy(); x["flow_m3d"] = Q[y].values * 86400.0
        a = analysis.analyse(x.flow_m3d.values, x.diameter_mm.values / 1000, s_new, nom.values)
        x["velocity_ms"] = a.velocity_ms.values
        x["depth_in_m"] = x["depth_out_m"] = a.d_over_D.values * x.diameter_mm.values / 1000
        x.to_csv(os.path.join(EXP, f), index=False)
    shutil.copy(os.path.join(INP, "manholes_network.csv"), os.path.join(EXP, f"manholes_{tag}a.csv"))
    pd.DataFrame(rows_out, columns=["year", "source", "receiver_id", "receiver_label", "flow_m3d"]).to_csv(
        os.path.join(EXP, f"transfers_{tag}.csv"), index=False)
    pd.DataFrame(lift_rows, columns=["year", "cut_pipe", "receiver_id", "flow_m3d"]).to_csv(
        os.path.join(EXP, f"lift_transfers_{tag}.csv"), index=False)
    S = dict(tag=tag, scenario=ibri_routing.SCENARIOS[name]["text"], passes=passes, pipes=len(d),
             over_largest_size=int(over.sum()), sizes={int(k): int(v) for k, v in nom.value_counts().sort_index().items()},
             km_above_dn200=round(float(d.length_m[nom.values > 200].sum()) / 1000, 2),
             largest=int(nom.max()), q_design_max_ls=round(float(Q[year].max()) * 1000, 1))
    json.dump(S, open(os.path.join(R, tag, "design.json"), "w"), indent=1)
    return S


if __name__ == "__main__":
    a = [x for x in sys.argv[1:] if not x.startswith("--")]
```

```python
def engineering_differences(case_dir, case):
    """The spec's phase-1 tolerances (section 7): identical sizes, depths +-1 mm, flows 0.1 %, the same stations at the
    same manholes, the same km under 0.75 m/s at 2031 and 2036 (+-0.01 km), the same deepest manhole (+-1 mm)."""
    out = []
    a = pd.read_csv(os.path.join(case_dir, "named", "links.csv")).set_index("key")
    b = fixtures.read_csv(f"golden/{case}/named/links.csv").set_index("key")
    if not a.index.equals(b.index): return [f"links: different pipe keys ({len(a)} vs {len(b)})"]
    n = int((a.dn_mm != b.dn_mm).sum())
    if n: out.append(f"size: {n} pipes differ, first {a.index[a.dn_mm != b.dn_mm][0]}")
    for col in ("depth_up_m", "depth_dn_m"):
        n = int(((a[col] - b[col]).abs() > 0.001 + 1e-9).sum())
        if n: out.append(f"{col}: {n} pipes beyond 1 mm")
    qa = pd.read_csv(os.path.join(case_dir, "exports", f"conduits_{case}a.csv")).set_index("label").flow_m3d
    qb = fixtures.read_csv(f"golden/{case}/exports/conduits_{case}a.csv").set_index("label").flow_m3d
    rel = ((qa - qb.reindex(qa.index)).abs() / qb.reindex(qa.index)).fillna(0)
    if (rel > 0.001).any(): out.append(f"flow: {int((rel > 0.001).sum())} pipes beyond 0.1 %, largest {rel.max() * 100:.3f} %")
    cols = ["name", "old", "kind", "system", "pumps_to"]
    sa = pd.read_csv(os.path.join(case_dir, "named", "stations.csv"))[cols]
    sb = fixtures.read_csv(f"golden/{case}/named/stations.csv")[cols]
    if not sa.equals(sb): out.append("stations: the list differs")
    ja = json.load(open(os.path.join(case_dir, "products", "quantities.json")))
    jb = json.load(open(fixtures.path(f"golden/{case}/products/quantities.json")))
    for k in ("km_big_under075_2031", "km_big_under075_2036"):
        if abs(ja[k] - jb[k]) > 0.01: out.append(f"{k}: {ja[k]} vs {jb[k]}")
    if ja["deepest"]["name"] != jb["deepest"]["name"] or abs(ja["deepest"]["depth"] - jb["deepest"]["depth"]) > 0.001:
        out.append(f"deepest: {ja['deepest']} vs {jb['deepest']}")
    return out
```

```python
    own = pd.read_csv(os.path.join(case_dir, "named", "links.csv")).set_index("new")
    q_own = pd.read_csv(os.path.join(case_dir, "exports", f"conduits_{case}a.csv")).set_index("label").flow_m3d
    own["q_own"] = own.old.map(q_own)
    sg = fixtures.read_csv(f"sewergems/{case}/conduits_{case}.csv"); sg = sg[sg.active == True].set_index("label")

           .join(s31[["velocity_ms"]].rename(columns={"velocity_ms": "v31_sg"}))
    dq = (j.flow_m3d - j.q_own) / j.q_own * 100

                over_limit=int((dD > lim + 0.005).sum()),
                dv_p05=float(dv.quantile(.05)), dv_p95=float(dv.quantile(.95)),
                km_under_model=float(big.length_m[big.v31_sg.round(2) < 0.75].sum() / 1000),
                km_under_own=float(big.length_m[big.v2031_ms.round(2) < 0.75].sum() / 1000))
```
